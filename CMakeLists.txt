cmake_minimum_required(VERSION 3.28...3.30)
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/common/bootstrap.cmake" NO_POLICY_SCOPE)
project(${_name} VERSION ${_version})
option(ENABLE_FRONTEND_API "Use OBS tools menu" ON)
option(ENABLE_QT "Use Qt" OFF)
include(compilerconfig)
include(defaults)
include(helpers)
add_library(${CMAKE_PROJECT_NAME} MODULE)
find_package(libobs REQUIRED)
find_package(obs-frontend-api REQUIRED)
target_link_libraries(${CMAKE_PROJECT_NAME} PRIVATE OBS::libobs OBS::obs-frontend-api)
target_sources(${CMAKE_PROJECT_NAME} PRIVATE src/plugin-main.cpp src/service.cpp src/input.cpp src/http.cpp)
target_include_directories(${CMAKE_PROJECT_NAME} PRIVATE third_party)
target_compile_features(${CMAKE_PROJECT_NAME} PRIVATE cxx_std_17)
if(WIN32)
  target_compile_definitions(${CMAKE_PROJECT_NAME} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(${CMAKE_PROJECT_NAME} PRIVATE ws2_32 xinput9_1_0 dinput8 dxguid ole32 user32 shell32)
endif()
install(FILES web/input.html DESTINATION "${CMAKE_PROJECT_NAME}/data/web")
install(FILES LICENSE NOTICE.md third_party/nlohmann/LICENSE.MIT DESTINATION "${CMAKE_PROJECT_NAME}/data/licenses")
set_target_properties_plugin(${CMAKE_PROJECT_NAME} PROPERTIES OUTPUT_NAME ${_name})
