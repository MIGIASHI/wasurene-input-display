cmake_minimum_required(VERSION 3.20)
project(wasurene_input_display VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
find_package(Threads REQUIRED)
add_library(input_core src/input.cpp src/http.cpp)
target_include_directories(input_core PUBLIC src third_party)
target_link_libraries(input_core PUBLIC Threads::Threads)
if(WIN32)
  target_compile_definitions(input_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
  target_link_libraries(input_core PUBLIC ws2_32 xinput9_1_0 dinput8 dxguid ole32 user32)
endif()
if(MSVC)
  target_compile_options(input_core PUBLIC /utf-8)
endif()
add_executable(wasurene-input-display src/main.cpp)
target_link_libraries(wasurene-input-display PRIVATE input_core)
if(WIN32)
  set_target_properties(wasurene-input-display PROPERTIES WIN32_EXECUTABLE TRUE)
  target_link_libraries(wasurene-input-display PRIVATE shell32)
endif()
add_custom_command(TARGET wasurene-input-display POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/web" "$<TARGET_FILE_DIR:wasurene-input-display>/web")
add_executable(input_check tests/input_check.cpp)
target_link_libraries(input_check PRIVATE input_core)
add_executable(http_check tests/http_check.cpp)
target_link_libraries(http_check PRIVATE input_core)
