# Shared between the two WASURENE OBS plugins. No external updater application is required.
enable_language(C)
option(WASURENE_UPDATER_TESTING "Include sandbox and dialog tests (never publish as stable)" OFF)
add_library(wasurene_update_core STATIC core.cpp vendor/miniz/miniz.c vendor/miniz/miniz_tdef.c vendor/miniz/miniz_tinfl.c vendor/miniz/miniz_zip.c)
target_include_directories(wasurene_update_core PUBLIC . ../third_party PRIVATE vendor/miniz)
target_compile_features(wasurene_update_core PUBLIC cxx_std_17)
target_compile_definitions(wasurene_update_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_link_libraries(wasurene_update_core PUBLIC winhttp bcrypt)
add_executable(wasurene-update WIN32 helper.cpp)
target_link_libraries(wasurene-update PRIVATE wasurene_update_core shell32)
target_compile_definitions(wasurene-update PRIVATE $<$<BOOL:${WASURENE_UPDATER_TESTING}>:WASURENE_UPDATER_TESTING>)
if(MSVC)
 target_compile_options(wasurene_update_core PRIVATE /utf-8)
 target_compile_options(wasurene-update PRIVATE /utf-8)
endif()
install(TARGETS wasurene-update RUNTIME DESTINATION "${CMAKE_PROJECT_NAME}/data/updater")
install(FILES README.md DESTINATION "${CMAKE_PROJECT_NAME}/data/updater")
install(FILES vendor/miniz/LICENSE DESTINATION "${CMAKE_PROJECT_NAME}/data/licenses" RENAME miniz-LICENSE.txt)
