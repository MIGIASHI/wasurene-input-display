cmake_minimum_required(VERSION 3.20)
project(input_display_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Threads REQUIRED)
add_library(input_core ../src/input.cpp ../src/http.cpp ../src/service.cpp)
target_include_directories(input_core PUBLIC ../src ../third_party)
target_link_libraries(input_core PUBLIC Threads::Threads)
if(WIN32)
  target_compile_definitions(input_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
  target_link_libraries(input_core PUBLIC ws2_32 xinput9_1_0 dinput8 dxguid ole32 user32)
endif()
if(MSVC)
  target_compile_options(input_core PUBLIC /utf-8)
endif()
add_executable(service_check service_check.cpp)
target_link_libraries(service_check PRIVATE input_core)
add_custom_command(TARGET service_check POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
  "${CMAKE_CURRENT_SOURCE_DIR}/../web" "$<TARGET_FILE_DIR:service_check>/web")
add_executable(input_check input_check.cpp)
target_link_libraries(input_check PRIVATE input_core)
add_executable(http_check http_check.cpp)
target_link_libraries(http_check PRIVATE input_core)
